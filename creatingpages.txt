CREATE lib/api.ts:
- Axios instance with baseURL from NEXT_PUBLIC_API_URL
- Request interceptor: attach Authorization: Bearer {token} from localStorage
- Response interceptor: on 401 → redirect to /login
- Export typed api.get(), api.post(), api.patch(), api.delete()

CREATE lib/utils.ts:
- cn() function using clsx + tailwind-merge
- formatDate(date) → "Jan 15, 2025"
- formatRelativeTime(date) → "2 hours ago"
- getStatusColor(status) → returns Tailwind color class
- getPriorityColor(priority) → returns color class
- getRiskColor(score) → green/yellow/red based on 0-100 score
- getMeritColor(score) → color for merit score badges

CREATE tailwind.config.ts:
- Custom color palette:
  primary: { 50-950 using #2563EB as base }
  dark: { DEFAULT: #0D1B2A, card: #1B3A5C }
  success: #10B981
  warning: #F59E0B
  danger: #EF4444
  cyan: #06B6D4
- Custom font: Inter from Google Fonts
- Custom animation: fade-in, slide-up, pulse-slow

name: Deploy GitHub Pages
on:
  push:
    branches: [main]
    paths: ['docs/**']
jobs:
  deploy:
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4
      - name: Deploy to GitHub Pages
        uses: peaceiris/actions-gh-pages@v3
        with:
          github_token: ${{ secrets.GITHUB_TOKEN }}
          publish_dir: ./docs
